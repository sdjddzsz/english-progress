```sql
-- 1) 新列（词本轮第一次露面时间；首答写一次不覆盖；老数据回落本机补）
alter table book_words add column if not exists "firstSeenAt" timestamptz;

-- 2) 计划卡聚合（一参！函数体自己算 14 天窗口，不需要 p_today）
create or replace function ep_home_summary(p_book_id text) returns json language sql stable as $fn$
  select json_build_object(
    'remaining', (select count(*) from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and b.stage > 0),
    'graduated', (select count(*) from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and b.stage = 0),
    'dailyCounts', coalesce((select json_agg(json_build_object('date', d, 'count', c) order by d) from (select (b."lastQuizAt" at time zone 'UTC')::date as d, count(distinct b."wordId") as c from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and b."lastQuizAt" is not null and b."lastQuizAt" >= (now() - interval '14 days') group by 1) t), '[]'::json),
    'tierCounts', coalesce((select json_object_agg(t.tier, t.c) from (select b.tier as tier, count(*) as c from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() group by 1) t), '{}'::json),
    'wordsByTier', coalesce((select json_object_agg(t.tier, t.ids) from (select b.tier as tier, json_agg(b."wordId" order by b."listNo", b."orderInList", b.id) as ids from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() group by 1) t), '{}'::json)
  ) $fn$

-- 3) 今日检测聚合
create or replace function ep_today_quiz(p_book_id text, p_today date) returns json language sql stable as $fn$
  select json_build_object(
    'words', (select count(*) from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and (b."lastQuizAt" at time zone 'UTC')::date = p_today),
    'firstSeen', (select count(*) from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and (b."firstSeenAt" at time zone 'UTC')::date = p_today),
    'wordIds', coalesce((select json_agg(b."wordId") from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and (b."lastQuizAt" at time zone 'UTC')::date = p_today), '[]'::json),
    'firstSeenIds', coalesce((select json_agg(b."wordId") from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and (b."firstSeenAt" at time zone 'UTC')::date = p_today), '[]'::json)
  ) $fn$

-- 4) 首页四阶段卡片聚合
create or replace function ep_stage_stats(p_book_id text) returns json language sql stable as $fn$
  select json_build_object(
    'byStage', coalesce((select json_object_agg(t.stage, t.c) from (select b.stage as stage, count(*) as c from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() group by 1) t), '{}'::json),
    'specialByStage', coalesce((select json_object_agg(t.stage, t.c) from (select b.stage as stage, count(*) as c from book_words b where b."bookId"=p_book_id and b."owner_id"=auth.uid() and b."specialTraining" group by 1) t), '{}'::json)
  ) $fn$
```

## ★ 本次踩的三个坑（下一个会话必读）

1. **RPC 签名不匹配 = 上线即 404**：云端 `ep_home_summary` 最初建成 `(p_book_id text, p_today date)` **两参**（函数体根本不用 p_today），而应用只传 `p_book_id` **一个参数**。上线后首页统计全挂（404 → repo 抛错 → 计划卡报错）。**已 drop 两参版重建一参版**，并用 `pg_proc.proargnames` 逐个核对三个函数签名与应用调用一致。⇒ 以后每次动云函数签名，必须 `pg_proc` 核对，别信「建过就是对的」。
2. **真 SDK 的 rpc() 回的是信封，不是裸值**：`await db.rpc(...)` 得到 `processResponse` 的 `{success, error, data, count, status, statusText}`。wordRepo 最初直接 cast 成业务类型读字段 ⇒ 线上统计全 0 且不报错——**是 MockDb 回裸对象的桩把缺陷兜住了**（测试全绿）。已修：repo 走 `rpcData` 拆信封、`CloudDatabaseLike.rpc` 类型收紧、mock 改回信封形状、未知函数回 404 错误信封（不 throw）。
3. **云 exec_sql 两个硬限制**：① prepared statement 一次只能一条命令（`drop function if exists a(); drop function if exists b();` 直接报错）；② 单条 SQL ~1200 字符上限（~1900 报 `syntax error at end of input`）⇒ 聚合必须拆函数（这也是三个函数而不是一个的原因）；③ 长 SQL 的 dollar-quoting 用自定义标签 `$fn$`（`$$` 在长语句上失败）。
   - anon key 直调 `/rpc/` 返回 401 invalid_client（需用户态 Bearer）⇒ RPC 网关只能靠 SDK 源码取证 + 真云建函数 read 模式验证；回退预案（SQL 视图走 `from('v_...')`）未用上。

`words.word_fine` 的 `sensesDetail` 中 **sense#1/#3/#5** 三条义项音标被误植成 `/ˈfɪ.nə/`，修正 SQL（保序 jsonb_agg + jsonb_set，数组长度不变）：

```sql
update words set "sensesDetail" = (select jsonb_agg(el2 order by ord) from (select ord, case when el->>'senseId' in ('fine#1','fine#3','fine#5') then jsonb_set(jsonb_set(jsonb_set(el, '{ipa}', '"/ˈfaɪn/"'), '{arpabet}', '"F AY1 N"'), '{pronunciationKey}', '"F AY1 N"') else el end as el2 from jsonb_array_elements("sensesDetail") with ordinality as t(el, ord)) s) where id = 'word_fine'
```

### 为什么不能只改 IPA（设计根子）

TTS 双轨制：**DashScope 主轨用 ARPAbet**（`<phoneme alphabet="cmu">`），Azure 轨才用 IPA。只改 IPA 不改 ARPAbet ⇒ 显示修好、音频仍错（fine 三个义项正是两边都错）。所以音标编辑只让用户改 IPA，**ARPAbet 由 `ipaToArpabet` 逆向派生**并在弹窗明示；`pronunciationKey` 随新 ARPAbet 重算（读音分组的依据，明示影响）。
